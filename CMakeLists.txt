cmake_minimum_required(VERSION 3.16)
# ESP-IDF and the host test build both read this release identifier.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/version.txt")
if(GD_HOST)
    project(GoldDiggerHost C)
    set(CMAKE_C_STANDARD 11)
    enable_testing()
    add_library(gd_core core/digger.c)
    target_include_directories(gd_core PUBLIC core)
    add_executable(core_tests tests/core_tests.c)
    target_link_libraries(core_tests gd_core)
    add_test(NAME core_tests COMMAND core_tests)
else()
    file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/version.txt" PROJECT_VER LIMIT_COUNT 1)
    set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/platforms/passport/components" "${CMAKE_CURRENT_LIST_DIR}/platforms/passport/main")
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(GoldDigger)
endif()
